int result = kill(pid_t pid, int sig);
    int sig: The signal that should be sent to the processes specified by pid. This must be 0 or one of the signals defined in the signal.h header file. If sig is 0, kill() performs error checking but does not send a signal. You can code sig as 0 to check whether the pid argument is valid. 
    
Returned value
kill() returns 0 if it has permission to send sig to any of the processes specified by pid. If kill() fails to send a signal, it returns -1 and sets errno to one of the following values:
-> EINVAL: The value of sig is incorrect or is not the number of a supported signal.
-> EPERM: The caller does not have permission to send the signal to any process specified by pid.
-> ESRCH: There are no processes or process groups corresponding to pid.

char **lsh_parse_line(char *line, int *is_background) {
    int bufsize = MAX_ARGS, position = 0;
    char **tokens = malloc(bufsize * sizeof(char *));
    char *token;

    if (!tokens) {
        fprintf(stderr, "lsh: allocation error\n");
        exit(EXIT_FAILURE);
    }

    token = strtok(line, " \t\r\n");
    while (token != NULL) {
        if (strcmp(token, "&") == 0) {
            *is_background = 1; // Proces w tle
            break;
        }

        tokens[position++] = token;

        if (position >= bufsize) {
            bufsize += MAX_ARGS;
            tokens = realloc(tokens, bufsize * sizeof(char *));
            if (!tokens) {
                fprintf(stderr, "lsh: allocation error\n");
                exit(EXIT_FAILURE);
            }
        }

        token = strtok(NULL, " \t\r\n");
    }
    tokens[position] = NULL;
    return tokens;
}

lsh_parse_line: Funkcja parsująca linię wejściową na argumenty.

    *char line: Linia wejściowa.

    *int is_background: Zmienna wskazująca, czy proces ma być uruchomiony w tle.

    Inicjalizacja zmiennych:

        int bufsize = MAX_ARGS, position = 0: Inicjalizacja rozmiaru bufora i pozycji w tablicy tokenów.

        **char *tokens = malloc(bufsize * sizeof(char )): Alokacja pamięci na tablicę tokenów.

        *char token: Zmienna przechowująca pojedynczy token.

    Sprawdzenie alokacji pamięci:

        if (!tokens): Sprawdza, czy alokacja pamięci się powiodła.

        fprintf(stderr, "lsh: allocation error\n"); exit(EXIT_FAILURE): Wyświetla błąd i kończy program, jeśli alokacja się nie powiodła.

    Tokenizacja linii:

        token = strtok(line, " \t\r\n"): Dzieli linię na tokeny za pomocą znaków rozdzielających (spacja, tabulator, powrót karetki, nowa linia).

        while (token != NULL): Pętla przechodząca przez wszystkie tokeny.

    Sprawdzenie procesu w tle:

        if (strcmp(token, "&") == 0): Sprawdza, czy token to "&".

        is_background = 1: Ustawia zmienną is_background na 1, oznaczając proces w tle.

        break: Przerywa pętlę.

    Dodawanie tokenów do tablicy:

        tokens[position++] = token: Dodaje token do tablicy tokens.

        if (position >= bufsize): Sprawdza, czy tablica tokenów jest pełna.

        bufsize += MAX_ARGS: Zwiększa rozmiar bufora.

        *tokens = realloc(tokens, bufsize * sizeof(char )): Reallokuje pamięć dla tablicy tokenów.

        if (!tokens): Sprawdza, czy reallokacja pamięci się powiodła.

int lsh_handle_builtins(char **args) {
    if (args[0] == NULL) {
        return 1;
    }
    if (strcmp(args[0], "exit") == 0) {
        exit(0);
    } else if (strcmp(args[0], "cd") == 0) {
        if (args[1] == NULL) {
            fprintf(stderr, "lsh: expected argument to \"cd\"\n");
        } else if (chdir(args[1]) != 0) {
            perror("lsh");
        }
        return 1;
    }
    return 0;
}

    **char args: Tablica argumentów.

    Sprawdzenie pustych argumentów:

        if (args[0] == NULL): Sprawdza, czy pierwszy argument jest pusty.

        return 1: Zwraca 1, co oznacza, że komenda została przetworzona (brak komendy do wykonania).

    Obsługa komendy "exit":

        if (strcmp(args[0], "exit") == 0): Sprawdza, czy pierwszy argument to "exit".

        exit(0): Kończy program z kodem wyjścia 0.

    Obsługa komendy "cd":

        else if (strcmp(args[0], "cd") == 0): Sprawdza, czy pierwszy argument to "cd".

        if (args[1] == NULL): Sprawdza, czy drugi argument (ścieżka) jest pusty.

        fprintf(stderr, "lsh: expected argument to \"cd\"\n"): Wyświetla komunikat o błędzie, jeśli brakuje argumentu.

Wykonanie pojedynczej komendy
c

int lsh_execute_command(char **args, int in_fd, int out_fd, int err_fd, int is_background) {
    pid_t pid;
    int status;

    pid = fork();
    if (pid == 0) {
        // Proces potomny
        if (in_fd != STDIN_FILENO) {
            dup2(in_fd, STDIN_FILENO);
            close(in_fd);
        }
        if (out_fd != STDOUT_FILENO) {
            dup2(out_fd, STDOUT_FILENO);
            close(out_fd);
        }
        if (err_fd != STDERR_FILENO) {
            dup2(err_fd, STDERR_FILENO);
            close(err_fd);
        }
        if (execvp(args[0], args) == -1) {
            perror("lsh");
        }
        exit(EXIT_FAILURE);
    } else if (pid < 0) {
        // Błąd forkowania
        perror("lsh");
    } else {
        if (is_background) {
            // Proces w tle - nie czekaj na zakończenie
            printf("[Proces w tle uruchomiony] PID: %d\n", pid);
        } else {
            // Proces pierwszoplanowy
            foreground_process = pid;
            waitpid(pid, &status, 0);
            foreground_process = 0;
        }
    }

    return 1;
}

lsh_execute_command: Funkcja wykonująca pojedynczą komendę, uwzględniając przekierowania i procesy w tle.

    **char args: Tablica argumentów.

    int in_fd: Deskryptor pliku wejściowego.

    int out_fd: Deskryptor pliku wyjściowego.

    int err_fd: Deskryptor pliku wyjścia błędów.

    int is_background: Flaga wskazująca, czy proces ma być uruchomiony w tle.

    Deklaracja zmiennych:

        pid_t pid: Identyfikator procesu.

        int status: Status zakończenia procesu.

    Forkowanie procesu:

        pid = fork(): Tworzy nowy proces potomny.

    Proces potomny:

        if (pid == 0): Kod wykonywany przez proces potomny.

            if (in_fd != STDIN_FILENO): Sprawdza, czy wejście jest przekierowane.

                dup2(in_fd, STDIN_FILENO): Duplikuje deskryptor pliku na standardowe wejście.

                close(in_fd): Zamknięcie oryginalnego deskryptora pliku.

            if (out_fd != STDOUT_FILENO): Sprawdza, czy wyjście jest przekierowane.

                dup2(out_fd, STDOUT_FILENO): Duplikuje deskryptor pliku na standardowe wyjście.

                close(out_fd): Zamknięcie oryginalnego deskryptora pliku.

            if (err_fd != STDERR_FILENO): Sprawdza, czy wyjście błędów jest przekierowane.

                dup2(err_fd, STDERR_FILENO): Duplikuje deskryptor pliku na standardowe wyjście błędów.

                close(err_fd): Zamknięcie oryginalnego deskryptora pliku.

            execvp(args[0], args): Uruchamia nowy program wskazany przez args[0] z argumentami args.

                if (execvp(args[0], args) == -1): Sprawdza, czy uruchomienie programu się powiodło.

                    perror("lsh"): Wyświetla komunikat o błędzie, jeśli uruchomienie programu się nie powiodło.

                    exit(EXIT_FAILURE): Kończy proces potomny z kodem wyjścia EXIT_FAILURE.

    Błąd forkowania:

        else if (pid < 0): Sprawdza, czy forkowanie się nie powiodło.

            perror("lsh"): Wyświetla komunikat o błędzie.

    Proces rodzicielski:

        else: Kod wykonywany przez proces rodzicielski.

            if (is_background): Sprawdza, czy proces ma być uruchomiony w tle.

                printf("[Proces w tle uruchomiony] PID: %d\n", pid): Wyświetla komunikat o uruchomieniu procesu w tle.

            else: Proces pierwszoplanowy.

                foreground_process = pid: Ustawia zmienną globalną na PID procesu pierwszoplanowego.

                waitpid(pid, &status, 0): Czeka na zakończenie procesu potomnego.

                foreground_process = 0: Resetuje zmienną globalną.

Wykonanie komendy z obsługą potoków i przekierowań (kontynuacja)
c

int lsh_execute(char **args, int is_background) {
    int in_fd = STDIN_FILENO;
    int out_fd = STDOUT_FILENO;
    int err_fd = STDERR_FILENO;

    char *cmd[MAX_ARGS];
    int cmd_index = 0;

    for (int i = 0; args[i] != NULL; i++) {
        if (strcmp(args[i], "|") == 0) {
            // Potoki
            int pipe_fd[2];
            pipe(pipe_fd);

            cmd[cmd_index] = NULL;
            lsh_execute_command(cmd, in_fd, pipe_fd[1], err_fd, 0);

            close(pipe_fd[1]);
            in_fd = pipe_fd[0];
            cmd_index = 0;
        } else if (strcmp(args[i], "<") == 0) {
            // Przekierowanie wejścia
            in_fd = open(args[++i], O_RDONLY);
            if (in_fd < 0) {
                perror("lsh");
                return 1;
            }
        } else if (strcmp(args[i], ">") == 0) {
            // Przekierowanie wyjścia
            out_fd = open(args[++i], O_WRONLY | O_CREAT | O_TRUNC, 0644);
            if (out_fd < 0) {
                perror("lsh");
                return 1;
            }
        } else if (strcmp(args[i], "2>") == 0) {
            // Przekierowanie wyjścia błędu
            err_fd = open(args[++i], O_WRONLY | O_CREAT | O_TRUNC, 0644);
            if (err_fd < 0) {
                perror("lsh");
                return 1;
            }
        } else {
            // Dodanie do aktualnej komendy
            cmd[cmd_index++] = args[i];
        }
    }

    cmd[cmd_index] = NULL;
    lsh_execute_command(cmd, in_fd, out_fd, err_fd, is_background);

    if (in_fd != STDIN_FILENO) close(in_fd);
    if (out_fd != STDOUT_FILENO) close(out_fd);
    if (err_fd != STDERR_FILENO) close(err_fd);

    return 1;
}

lsh_execute: Funkcja wykonująca komendę z obsługą potoków i przekierowań.

    **char args: Tablica argumentów.

    int is_background: Flaga wskazująca, czy proces ma być uruchomiony w tle.

    Inicjalizacja deskryptorów plików:

        int in_fd = STDIN_FILENO: Ustawia deskryptor pliku wejściowego na standardowe wejście.

        int out_fd = STDOUT_FILENO: Ustawia deskryptor pliku wyjściowego na standardowe wyjście.

        int err_fd = STDERR_FILENO: Ustawia deskryptor pliku wyjścia błędów na standardowe wyjście błędów.

    Inicjalizacja tablicy komend:

        *char cmd[MAX_ARGS]: Tablica do przechowywania aktualnej komendy.

        int cmd_index = 0: Indeks do przechowywania bieżącej pozycji w tablicy cmd.

    Iteracja przez argumenty:

        for (int i = 0; args[i] != NULL; i++): Przechodzi przez wszystkie argumenty.

    Obsługa potoków:

        if (strcmp(args[i], "|") == 0): Sprawdza, czy argument to "|".

            int pipe_fd[2]: Tablica do przechowywania deskryptorów plików potoku.

            pipe(pipe_fd): Tworzy potok.

            cmd[cmd_index] = NULL: Ustawia koniec tablicy komend.

            lsh_execute_command(cmd, in_fd, pipe_fd[1], err_fd, 0): Wykonuje aktualną komendę z wyjściem skierowanym do potoku.

            close(pipe_fd[1]): Zamykacie końcówkę wyjściową potoku.

            in_fd = pipe_fd[0]: Ustawia deskryptor pliku wejściowego na końcówkę wejściową potoku.

            cmd_index = 0: Resetuje indeks tablicy komend.

    Obsługa przekierowania wejścia:

        else if (strcmp(args[i], "<") == 0): Sprawdza, czy argument to "<".

            in_fd = open(args[++i], O_RDONLY): Otwiera plik do odczytu i ustawia deskryptor pliku wejściowego.

            if (in_fd < 0): Sprawdza, czy otwarcie pliku się powiodło.

                perror("lsh"): Wyświetla komunikat o błędzie, jeśli otwarcie pliku się nie powiodło.

                return 1: Zwraca 1, co oznacza błąd.

    Obsługa przekierowania wyjścia:

        else if (strcmp(args[i], ">") == 0): Sprawdza, czy argument to ">".

            out_fd = open(args[++i], O_WRONLY | O_CREAT | O_TRUNC, 0644): Otwiera plik do zapisu, tworzy go, jeśli nie istnieje, i ustawia deskryptor pliku wyjściowego.

            if (out_fd < 0): Sprawdza, czy otwarcie pliku się powiodło.

                perror("lsh"): Wyświetla komunikat o błędzie, jeśli otwarcie pliku się nie powiodło.

                return 1: Zwraca 1, co oznacza błąd.

    Obsługa przekierowania wyjścia błędów:

        else if (strcmp(args[i], "2>") == 0): Sprawdza, czy argument to "2>".

            err_fd = open(args[++i], O_WRONLY | O_CREAT | O_TRUNC, 0644): Otwiera plik do zapisu błędów, tworzy go, jeśli nie istnieje, i ustawia deskryptor pliku wyjścia błędów.

            if (err_fd < 0): Sprawdza, czy otwarcie pliku się powiodło.

                perror("lsh"): Wyświetla komunikat o błędzie, jeśli otwarcie pliku się nie powiodło.

                return 1: Zwraca 1, co oznacza błąd.

    Dodanie argumentu do komendy:

        else: Jeśli argument nie jest specjalnym operatorem.

            cmd[cmd_index++] = args[i]: Dodaje argument do tablicy komend.

    Zakończenie tablicy komend:

        cmd[cmd_index] = NULL: Ustawia koniec tablicy komend.

    Wykonanie komendy:

        lsh_execute_command(cmd, in_fd, out_fd, err_fd, is_background): Wykonuje komendę z odpowiednimi przekierowaniami i flagą is_background.

    Zamykanie deskryptorów plików:

        if (in_fd != STDIN_FILENO) close(in_fd): Zamknięcie deskryptora pliku wejściowego, jeśli nie jest standardowym wejściem.

        if (out_fd != STDOUT_FILENO) close(out_fd): Zamknięcie deskryptora pliku wyjściowego, jeśli nie jest standardowym wyjściem.

        if (err_fd != STDERR_FILENO) close(err_fd): Zamknięcie deskryptora pliku wyjścia błędów, jeśli nie jest standardowym wyjściem błędów.

    Zwrócenie 1: Funkcja zawsze zwraca 1 (komenda została przetworzona).

void lsh_loop() {
    char *line = NULL;
    char **args;
    size_t bufsize = 0;

    signal(SIGCHLD, sigchld_handler);
    signal(SIGINT, sigint_handler);

    while (1) {
        printf("lsh> ");
        if (getline(&line, &bufsize, stdin) == -1) {
            printf("\n");
            break;
        }

        int is_background = 0;
        args = lsh_parse_line(line, &is_background);

        if (!lsh_handle_builtins(args)) {
            lsh_execute(args, is_background);
        }

        free(args);
    }

    free(line);
}

lsh_loop: Funkcja głównej pętli powłoki.

    *char line = NULL: Zmienna do przechowywania linii wejściowej.

    **char args: Tablica argumentów komendy.

    size_t bufsize = 0: Rozmiar bufora dla linii wejściowej.

    Rejestrowanie obsługi sygnałów:

        signal(SIGCHLD, sigchld_handler): Rejestruje funkcję obsługi sygnału SIGCHLD, która usunie procesy zombie.

        signal(SIGINT, sigint_handler): Rejestruje funkcję obsługi sygnału SIGINT, która przerywa tylko aktywny proces.

    Główna pętla powłoki:

        while (1): Nieskończona pętla powłoki.

    Wyświetlanie prompta:

        printf("lsh> "): Wyświetla prompt powłoki.

    Odczytywanie linii wejściowej:

        if (getline(&line, &bufsize, stdin) == -1): Odczytuje linię wejściową z terminala.

            getline: Funkcja standardowa do odczytu linii z wejścia.

            &line: Wskaźnik do zmiennej przechowującej linię wejściową.

            &bufsize: Wskaźnik do zmiennej przechowującej rozmiar bufora.

            stdin: Standardowe wejście (terminal).

            if (getline == -1): Sprawdza, czy odczyt linii się nie powiódł.

                printf("\n"): Wyświetla nową linię (domknięcie linii wejściowej).

                break: Przerywa pętlę (koniec powłoki).

    Parsowanie linii wejściowej na argumenty:

        int is_background = 0: Flaga wskazująca, czy proces ma być uruchomiony w tle.

        args = lsh_parse_line(line, &is_background): Parsuje linię wejściową na argumenty i ustawia flagę is_background.

    Obsługa wbudowanych komend:

        if (!lsh_handle_builtins(args)): Sprawdza, czy komenda jest wbudowana.

            lsh_handle_builtins: Funkcja obsługująca komendy wbudowane.

            if (!lsh_handle_builtins(args)): Jeśli komenda nie jest wbudowana, wykonuje ją za pomocą lsh_execute.

    Wykonanie komendy:

        lsh_execute(args, is_background): Wykonuje komendę z odpowiednimi przekierowaniami i flagą is_background.

    Zwalnianie pamięci:

        free(args): Zwalnia pamięć przydzieloną na tablicę argumentów.

    Zwalnianie pamięci na końcu pętli:

        free(line): Zwalnia pamięć przydzieloną na linię wejściową po zakończeniu pętli.
